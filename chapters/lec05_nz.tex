\section{Lecture 5: Nakajima--Zwanzig, and why the memory kernel is a self-energy}
\label{sec:lec5}

The Nakajima--Zwanzig (NZ) equation is an exact, closed equation of motion for
$\rhoS(t)$. It is derived in five lines with a projection operator, and its memory kernel
$K$ looks at first like a formal object. The content of this lecture is that $K$ is
exactly the sum of the irreducible diagrams of \cref{sec:lec4}, so that the NZ equation is
the Dyson equation of the real-time expansion. We then evaluate $K$ at second order for
the qubit and recover the Bloch--Redfield description, with every rate and shift expressed
through $S$ and $\chi$.

\subsection{The projection and the exact equation}
\label{sec:nz-derivation}

Define the Nakajima--Zwanzig projector~\cite{Nakajima1958,Zwanzig1960}
\begin{equation}
  \Pc X=\rhoB\otimes\Tr_B X,\qquad \Qc=1-\Pc,\qquad \Pc^2=\Pc .
  \label{eq:P}
\end{equation}
$\Pc$ keeps the ``relevant'' part, the reduced state tensored with the fixed reference bath
state. Three facts follow from $[H_B,\rhoB]=0$ and $\avg{B}=0$:
\begin{equation}
  \Pc\Lc_S=\Lc_S\Pc,\qquad \Pc\Lc_B=\Lc_B\Pc=0\ \text{on the range of }\Pc,\qquad \Pc\Lc_I\Pc=0 .
  \label{eq:P-facts}
\end{equation}
(The last one: $\Pc\Lc_I\Pc X=\rhoB\otimes\Tr_B[-i(S B\otimes\rhoB X'-\dots)]\propto\avg{B}=0$.)
Split the von Neumann equation $\dot\rho=\Lc\rho$ into its $\Pc$ and $\Qc$ components:
\begin{equation}
  \partial_t\Pc\rho=\Pc\Lc\Pc\rho+\Pc\Lc\Qc\rho,\qquad
  \partial_t\Qc\rho=\Qc\Lc\Pc\rho+\Qc\Lc\Qc\rho .
\end{equation}
The second is a linear inhomogeneous equation for $\Qc\rho$ with the known source $\Qc\Lc\Pc\rho(s)$:
\begin{equation}
  \Qc\rho(t)=\ee^{\Qc\Lc t}\Qc\rho(0)+\int_0^t\!\dd s\,\ee^{\Qc\Lc(t-s)}\,\Qc\Lc\Pc\rho(s).
\end{equation}
Insert into the first and use \cref{eq:P-facts}:
\begin{equation}
  \boxed{\;\dot\rhoS(t)=\Lc_S\rhoS(t)+\int_0^t\!\dd s\,K(t-s)\,\rhoS(s)+I(t),\qquad
  K(\tau)X=\Tr_B\!\Big[\Lc_I\,\ee^{\Qc\Lc\tau}\,\Qc\Lc_I\big(X\otimes\rhoB\big)\Big],\;}
  \label{eq:NZ}
\end{equation}
with the inhomogeneity $I(t)=\Tr_B[\Lc_I\ee^{\Qc\Lc t}\Qc\rho(0)]$, which vanishes for a
factorized initial state ($\Qc\rho(0)=0$). This is the NZ
equation~\cite[Sec.~9.1]{BreuerPetruccione}. It is exact. Nothing has been assumed about
the strength of the coupling, the Gaussianity of the bath or the size of the system. The
kernel $K(\tau)$ is a superoperator on the system; it depends on $\tau$ only (stationary
bath, time-independent $H$) and it contains the full $\Lc$ inside the exponential, which
is where all the difficulty hides.

\paragraph{Laplace form.}
With $\hat f(z)=\int_0^\infty\dd t\,\ee^{-zt}f(t)$ (Re$\,z>0$), the convolution becomes a product
and \cref{eq:NZ} for the map reads
\begin{equation}
  \boxed{\;\hat\Pi(z)=\frac{1}{z-\Lc_S-\hat K(z)}
  \quad\Longleftrightarrow\quad
  \hat\Pi=\hat\Pi_0+\hat\Pi_0\,\hat K\,\hat\Pi,\qquad \hat\Pi_0=(z-\Lc_S)^{-1}.\;}
  \label{eq:Dyson}
\end{equation}
This is a Dyson equation with $\hat K$ in the role of the self-energy. The next
subsection shows that the analogy is an identity.

\subsection{The kernel is the sum of irreducible diagrams}
\label{sec:nz-irreducible}

This subsection is the heart of the course, so we go slowly. The claim is that the
formal object $K(\tau)$ of \cref{eq:NZ}, with the full Liouvillian inside
$\ee^{\Qc\Lc\tau}$, is exactly the sum of the ``irreducible'' real-time diagrams of
\cref{sec:lec4}, and that the NZ equation is the Dyson resummation of the reducible
ones. We first fix the vocabulary, then prove three lemmas, then the theorem, then its
converse (the Dyson resummation), and finally check everything explicitly at fourth and
sixth order.

\subsubsection{Vocabulary: strings, gaps, cuts, pairings}

Throughout, $\Lc_0=\Lc_S+\Lc_B$ and $\tilde\Lc_I(t)=\ee^{-\Lc_0t}\Lc_I\ee^{\Lc_0t}$ is the interaction-picture
vertex, $\tilde\Lc_I(t)=-i[\tilde S^\qu(t)\tilde B^\cl(t)+\tilde S^\cl(t)\tilde B^\qu(t)]$ by \cref{eq:LIcq}.

\begin{definition}[Wick string]
For ordered times $t_0>t_1>\dots>t_{k}$ and a system operator $X$, the \emph{Wick
string} is the superoperator
\begin{equation}
  W(t_0,\dots,t_k)X\equiv\Tr_B\!\Big[\tilde\Lc_I(t_0)\tilde\Lc_I(t_1)\cdots\tilde\Lc_I(t_k)\,\big(X\otimes\rhoB\big)\Big].
  \label{eq:wick-string}
\end{equation}
Its $k$ \emph{gaps} are the intervals $(t_{j+1},t_j)$ between consecutive vertices,
$j=0,\dots,k-1$. By Wick's theorem \eqref{eq:wick} applied to the bath factors (every
vertex carries exactly one bath superoperator), $W$ is a sum over \emph{pairings}
$\pi$ of the $k+1$ vertices (so $k+1$ must be even, otherwise $W=0$):
\begin{equation}
  W(t_0,\dots,t_k)=\sum_{\pi}W_\pi(t_0,\dots,t_k),\qquad
  W_\pi=\prod_{\text{vertices}}\big(-i\,\tilde S^{a_j}(t_j)\big)\prod_{(ij)\in\pi}D^{\bar a_i\bar a_j}(t_i,t_j),
  \label{eq:wick-sum}
\end{equation}
where $a_j\in\{\cl,\qu\}$ is the system index at vertex $j$, $\bar a_j$ the opposite index,
and the sum over the $a_j$ is understood. $W_\pi$ is what we called a \emph{diagram}
in \cref{sec:lec4}; the Feynman rules there are just \cref{eq:wick-sum} with
\cref{eq:LO} inserted (recall $D^{\qu\qu}=0$ and $D^{\qu\cl}(t_i,t_j)=0$ for $t_i>t_j$, which
kills many assignments).
\end{definition}

\begin{definition}[Cuts, reducibility]
A pairing $\pi$ \emph{crosses} gap $j$ if some pair $(il)\in\pi$ has $i\le j<l$, i.e.\ one
vertex before the gap and one after it. A gap not crossed by $\pi$ is a \emph{cut} of
$\pi$. Write $U(\pi)\subseteq\{0,\dots,k-1\}$ for the set of cuts of $\pi$. A pairing is
\emph{irreducible} if $U(\pi)=\varnothing$ (every gap is crossed by at least one line) and
\emph{reducible} otherwise.
\end{definition}

At a cut, the vertices to the left and to the right are paired only among
themselves; the bath, which was in $\rhoB$ at the start, is ``back in $\rhoB$'' there in
the sense that no bath correlation connects the two sides. The first lemma makes this
precise.

\subsubsection{Three lemmas}

\begin{lemma}[A $\Pc$ in a gap factorizes the string]\label{lem:P-cut}
Inserting $\Pc$ in gap $j$ of a Wick string gives the product of the two sub-strings,
\begin{equation}
  \Tr_B\!\Big[\tilde\Lc_I(t_0)\cdots\tilde\Lc_I(t_j)\;\Pc\;\tilde\Lc_I(t_{j+1})\cdots\tilde\Lc_I(t_k)\,(X\otimes\rhoB)\Big]
  =W(t_0,\dots,t_j)\,W(t_{j+1},\dots,t_k)\,X ,
\end{equation}
and therefore equals $\sum_{\pi:\,j\in U(\pi)}W_\pi$, the sum over pairings that have a
cut at gap $j$. More generally, inserting $\Pc$ in every gap of a set $\mathcal C$ gives
$\sum_{\pi:\,\mathcal C\subseteq U(\pi)}W_\pi$.
\end{lemma}
\begin{proof}
By definition $\Pc Y=\rhoB\otimes\Tr_BY$. Apply it to
$Y=\tilde\Lc_I(t_{j+1})\cdots\tilde\Lc_I(t_k)(X\otimes\rhoB)$: $\Tr_BY=W(t_{j+1},\dots,t_k)X$ is a system
operator, and $\Pc Y=\big(W(t_{j+1},\dots,t_k)X\big)\otimes\rhoB$. The remaining string acts on
this factorized operator, which gives the first statement. For the second, apply Wick's
theorem to each factor: the left factor is a sum over pairings of $\{0,\dots,j\}$, the
right factor a sum over pairings of $\{j+1,\dots,k\}$, and their product is the sum over
pairings of $\{0,\dots,k\}$ that pair no vertex on the left with one on the right, i.e.\
those with a cut at $j$. The value of each such pairing is the same whether computed in
the full string or in the product, because the contraction $D^{ab}(t_i,t_l)$ of a pair
depends only on the two times and indices, and the time ordering is the same. The
statement for a set $\mathcal C$ follows by induction on $|\mathcal C|$.
\end{proof}

\begin{lemma}[Expansion of $\ee^{\Qc\Lc\tau}$]\label{lem:expansion}
With $\Pc$ of \cref{eq:P}, $[\Pc,\Lc_0]=0$, and for every $s$
\begin{equation}
  \ee^{\Qc\Lc_0s}=\Pc+\Qc\,\ee^{\Lc_0s},\qquad\text{hence}\qquad \ee^{\Qc\Lc_0s}\,\Qc=\Qc\,\ee^{\Lc_0s}=\ee^{\Lc_0s}\,\Qc .
  \label{eq:expQL0}
\end{equation}
Consequently
\begin{equation}
  \ee^{\Qc\Lc\tau}\,\Qc=\sum_{m\ge0}\int_{\tau>s_1>\dots>s_m>0}\!\!
  \Qc\,\ee^{\Lc_0(\tau-s_1)}\,\Qc\Lc_I\,\ee^{\Lc_0(s_1-s_2)}\,\Qc\Lc_I\cdots\ee^{\Lc_0(s_{m-1}-s_m)}\,\Qc\Lc_I\,\ee^{\Lc_0s_m}\,\Qc ,
  \label{eq:expQL}
\end{equation}
i.e.\ the ordinary Dyson series in $\Lc_I$ with free propagation $\ee^{\Lc_0\Delta s}$ between
vertices and a factor $\Qc$ in every gap (and at both ends).
\end{lemma}
\begin{proof}
$[\Pc,\Lc_S]=0$ because $\Lc_S$ acts on the system factor only:
$\Pc\Lc_SY=\rhoB\otimes\Tr_B(-i[H_S,Y])=-i[H_S,\rhoB\otimes\Tr_BY]=\Lc_S\Pc Y$. And
$\Pc\Lc_B=0$ (trace of a bath commutator) while $\Lc_B\Pc Y=-i[H_B,\rhoB]\otimes\Tr_BY=0$ (stationary
$\rhoB$); so $\Pc\Lc_B=\Lc_B\Pc=0$. Hence $\Qc$ commutes with $\Lc_0$ and
$(\Qc\Lc_0)^n=\Qc^n\Lc_0^n=\Qc\Lc_0^n$ for $n\ge1$, which gives
$\ee^{\Qc\Lc_0s}=1+\Qc(\ee^{\Lc_0s}-1)=\Pc+\Qc\ee^{\Lc_0s}$. Multiplying by $\Qc$ on the right and using
$\Pc\Qc=0$, $\Qc^2=\Qc$ gives \cref{eq:expQL0}. For \cref{eq:expQL}, write
$\Qc\Lc=\Qc\Lc_0+\Qc\Lc_I$ and expand $\ee^{(\Qc\Lc_0+\Qc\Lc_I)\tau}$ in the usual Dyson series with
``free'' part $\ee^{\Qc\Lc_0s}$ and ``perturbation'' $\Qc\Lc_I$; every $\ee^{\Qc\Lc_0\Delta s}$ is
adjacent to a $\Qc$ (from a neighbouring $\Qc\Lc_I$ or from the final $\Qc$), so
\cref{eq:expQL0} replaces it by $\ee^{\Lc_0\Delta s}$ with a $\Qc$ kept in the gap.
\end{proof}

\begin{lemma}[Inclusion--exclusion]\label{lem:incl-excl}
For any finite set $U$, $\sum_{\mathcal C\subseteq U}(-1)^{|\mathcal C|}=\delta_{U,\varnothing}$.
\end{lemma}
\begin{proof}
$\sum_{\mathcal C\subseteq U}(-1)^{|\mathcal C|}=\prod_{u\in U}(1-1)=0^{|U|}$, which is $1$ for $U=\varnothing$ and $0$ otherwise.
\end{proof}

\subsubsection{The theorem}

\begin{theorem}[NZ kernel $=$ irreducible diagrams]\label{thm:irreducible}
For the projector \eqref{eq:P} and a Gaussian bath with $\avg B=0$,
\begin{equation}
  K(\tau)=\ee^{\Lc_S\tau}\sum_{m\ge0}\int_{\tau>s_1>\dots>s_m>0}\ \sum_{\pi\ \mathrm{irreducible}}W_\pi(\tau,s_1,\dots,s_m,0),
  \label{eq:K-irred}
\end{equation}
where the inner sum runs over the irreducible pairings of the $m+2$ vertices at
$\tau,s_1,\dots,s_m,0$ (so only even $m$ contribute). In words: $K(\tau)$ is the sum of all
diagrams with outermost vertices at $0$ and $\tau$ that cannot be cut between two
consecutive vertices without cutting a line, dressed with the free system propagation
that brings the interaction-picture string back to the Schr\"odinger picture.
\end{theorem}

\begin{proof}
\emph{Step 1: reduce $K$ to a Wick string with $\Qc$'s in the gaps.} Insert
\cref{eq:expQL} into $K(\tau)X=\Tr_B[\Lc_I\ee^{\Qc\Lc\tau}\Qc\Lc_I(X\otimes\rhoB)]$. The two end
factors of $\Qc$ can be dropped: on the right, $\Qc\Lc_I\Pc=\Lc_I\Pc-\Pc\Lc_I\Pc=\Lc_I\Pc$ by
\cref{eq:P-facts}; on the left, $\Tr_B[\Lc_I\Qc Y]=\Tr_B[\Lc_IY]-\Tr_B[\Lc_I\Pc Y]$ and
$\Tr_B[\Lc_I(\rhoB\otimes Z)]=-i\Tr_B[(SB\otimes\rhoB Z-\dots)]\propto\avg B=0$. What remains is
\begin{equation}
  K(\tau)X=\sum_{m\ge0}\int_{\tau>s_1>\dots>s_m>0}\Tr_B\!\Big[\Lc_I\,\ee^{\Lc_0(\tau-s_1)}\,\Qc\Lc_I\,\ee^{\Lc_0(s_1-s_2)}\,\Qc\Lc_I\cdots\Qc\Lc_I\,\ee^{\Lc_0s_m}\,\Qc\,\Lc_I\,(X\otimes\rhoB)\Big].
\end{equation}
Note that there is one $\Qc$ in each of the $m+1$ gaps between the $m+2$ vertices (the
two $\Qc$'s that were adjacent in the first gap, $\Lc_I\Qc\ee^{\Lc_0(\tau-s_1)}\Qc\Lc_I$, merge into one
because $\Qc$ commutes with $\ee^{\Lc_0 s}$ and $\Qc^2=\Qc$; we then dropped the outer one).

\emph{Step 2: go to the interaction picture.} Write
$\ee^{\Lc_0(s_{j-1}-s_j)}=\ee^{\Lc_0s_{j-1}}\ee^{-\Lc_0s_j}$ and move all $\ee^{\pm\Lc_0s}$ next to the vertices,
using $[\Qc,\ee^{\Lc_0s}]=0$: the string becomes
$\ee^{\Lc_0\tau}\,\tilde\Lc_I(\tau)\,\Qc\,\tilde\Lc_I(s_1)\,\Qc\cdots\Qc\,\tilde\Lc_I(s_m)\,\Qc\,\tilde\Lc_I(0)$. The leftover
$\ee^{\Lc_0\tau}$ sits directly under $\Tr_B$, and $\Tr_B[\ee^{\Lc_0\tau}Y]=\ee^{\Lc_S\tau}\Tr_B[\ee^{\Lc_B\tau}Y]=\ee^{\Lc_S\tau}\Tr_BY$
because $\ee^{\Lc_B\tau}$ is a unitary conjugation on the bath factor, which the partial trace
does not see. Hence
\begin{equation}
  K(\tau)X=\ee^{\Lc_S\tau}\sum_{m\ge0}\int_{\tau>s_1>\dots>s_m>0}
  \Tr_B\!\Big[\tilde\Lc_I(\tau)\,\Qc\,\tilde\Lc_I(s_1)\,\Qc\,\tilde\Lc_I(s_2)\cdots\Qc\,\tilde\Lc_I(s_m)\,\Qc\,\tilde\Lc_I(0)\,(X\otimes\rhoB)\Big],
  \label{eq:K-Qstring}
\end{equation}
a Wick string of $m+2$ vertices with $\Qc=1-\Pc$ in each of its $m+1$ gaps.

\emph{Step 3: expand the $\Qc$'s.} $\prod_{j=0}^{m}(1-\Pc_j)=\sum_{\mathcal C\subseteq\{0,\dots,m\}}(-1)^{|\mathcal C|}\prod_{j\in\mathcal C}\Pc_j$,
where $\Pc_j$ denotes $\Pc$ inserted in gap $j$. By \cref{lem:P-cut}, the term with the
set $\mathcal C$ equals $\sum_{\pi:\,\mathcal C\subseteq U(\pi)}W_\pi$. Therefore
\begin{equation}
  \Tr_B\big[\cdots\Qc\cdots\Qc\cdots\big]
  =\sum_{\mathcal C}(-1)^{|\mathcal C|}\sum_{\pi:\,\mathcal C\subseteq U(\pi)}W_\pi
  =\sum_{\pi}W_\pi\sum_{\mathcal C\subseteq U(\pi)}(-1)^{|\mathcal C|}
  =\sum_{\pi}W_\pi\,\delta_{U(\pi),\varnothing},
\end{equation}
where we exchanged the two finite sums and used \cref{lem:incl-excl}. Only pairings
with no cut survive, which is \cref{eq:K-irred}.
\end{proof}

\begin{remark}
The three ingredients are worth separating. (i) $\Pc$ \emph{factorizes} a string
(\cref{lem:P-cut}); this uses only the form $\Pc=\rhoB\otimes\Tr_B$. (ii) The exponential
$\ee^{\Qc\Lc\tau}$ puts a $\Qc$ in every gap (\cref{lem:expansion}); this uses only
$[\Pc,\Lc_0]=0$. (iii) Inclusion--exclusion converts ``$\Qc$ in every gap'' into ``no cut
anywhere''; this is pure combinatorics. Gaussianity enters only in identifying the
factorized strings with sums over pairings. For a non-Gaussian bath the same three steps
go through with pairings replaced by partitions into cumulants, and the result is that
$K$ is the sum of diagrams in which every gap is crossed by at least one cumulant
block~\cite{Schoeller2009}. For fermionic reservoirs each $W_\pi$ acquires the sign of the
permutation that brings paired operators next to each other; nothing else changes.
\end{remark}

\subsubsection{The converse: the NZ equation as a Dyson resummation}

The theorem identifies $K$. The complementary statement is that the NZ equation
rebuilds the \emph{full} map from $K$ by summing the reducible diagrams, which is what one
means by a Dyson equation.

\begin{proposition}[Unique decomposition into irreducible blocks]\label{prop:blocks}
Every pairing $\pi$ of the vertices $t_0>t_1>\dots>t_k$ decomposes uniquely as a sequence
of irreducible pairings of consecutive blocks of vertices, the blocks being separated by
the cuts of $\pi$. Conversely, concatenating irreducible pairings of consecutive blocks
gives a pairing whose cuts are exactly the block boundaries.
\end{proposition}
\begin{proof}
Let $U(\pi)=\{j_1<\dots<j_r\}$. The vertices between consecutive cuts form blocks; no
line connects different blocks (that is what a cut means), so $\pi$ restricted to a
block is a pairing of that block; and that restriction has no cut, because a cut inside
a block would be a cut of $\pi$ not in $U(\pi)$. Uniqueness: the blocks are determined by
$U(\pi)$, which is determined by $\pi$. The converse is immediate.
\end{proof}

Now expand the full map $\Pi(t)=\Tr_B[\ee^{\Lc t}(\cdot\otimes\rhoB)]$ in the Dyson series
\eqref{eq:dyson-super} and apply Wick's theorem: in the Schr\"odinger picture, every
term is a product of free propagators $\ee^{\Lc_S\Delta t}$ and diagrams $W_\pi$. Group the
terms by the cuts of $\pi$ using \cref{prop:blocks}: a term with $r$ cuts is a product of
$r+1$ irreducible blocks, each of which is (after the time integrations over its interior
vertices) a factor $K(\tau_\ell)$ with $\tau_\ell$ the duration of the block, separated by free
propagation across the cuts. Summing over the number of blocks and over the positions of
the block boundaries,
\begin{equation}
  \Pi(t)=\Pi_0(t)+\int_0^t\!\dd t_1\!\int_0^{t_1}\!\dd t_2\,\Pi_0(t-t_1)K(t_1-t_2)\Pi_0(t_2)
  +\int\!\!\int\!\!\int\!\!\int\Pi_0K\Pi_0K\Pi_0+\dots ,\qquad \Pi_0(t)=\ee^{\Lc_St},
  \label{eq:dyson-time}
\end{equation}
which is the iterated solution of $\Pi(t)=\Pi_0(t)+\int_0^t\dd t_1\int_0^{t_1}\dd t_2\,\Pi_0(t-t_1)K(t_1-t_2)\Pi(t_2)$,
i.e.\ of the NZ equation $\dot\Pi=\Lc_S\Pi+\int K\,\Pi$. In Laplace space this is
\cref{eq:Dyson}, $\hat\Pi=(z-\Lc_S-\hat K)^{-1}=\hat\Pi_0+\hat\Pi_0\hat K\hat\Pi_0+\hat\Pi_0\hat K\hat\Pi_0\hat K\hat\Pi_0+\dots$, a
geometric series in $\hat K\hat\Pi_0$. The geometric series is precisely the statement that
reducible diagrams are products of irreducible ones with free propagation in between; the
time integrations over block boundaries turn the products into convolutions, and
convolutions into products of Laplace transforms. One point deserves emphasis: a given
term of the series \eqref{eq:dyson-time} with $r+1$ blocks is not ``of order $g^{2(r+1)}$''
in any useful sense, because each $K$ contains all orders. The expansion parameter of
\cref{eq:dyson-time} is the number of returns of the bath to $\rhoB$, not the coupling.

\subsubsection{Explicit checks at fourth and sixth order}

\begin{exbox}[Example 5.1: fourth order]
Take $m=2$ in \cref{eq:K-Qstring}: vertices at $\tau>s_1>s_2>0$, three gaps, and
$\prod_{j=0}^{2}(1-\Pc_j)$ has eight terms. Most vanish identically, for a reason that is
general: a $\Pc$ in gap $j$ splits the string into a left part with $j+1$ vertices and a
right part with $m+1-j$ vertices, and each part is a Wick string of its own, which
vanishes if it has an odd number of vertices. Hence $\Pc_0$ and $\Pc_2$ (one vertex on one
side) give zero, and so does any set $\mathcal C$ containing them. Only $\mathcal C=\varnothing$ and
$\mathcal C=\{1\}$ survive:
\begin{align}
  K^{(4)}(\tau)&=\ee^{\Lc_S\tau}\!\int_{\tau>s_1>s_2>0}\Big\{W(\tau,s_1,s_2,0)-W(\tau,s_1)\,W(s_2,0)\Big\}\notag\\
  &=\ee^{\Lc_S\tau}\!\int_{\tau>s_1>s_2>0}\Big\{\big[W_{(\tau s_1)(s_20)}+W_{(\tau s_2)(s_10)}+W_{(\tau0)(s_1s_2)}\big]-W_{(\tau s_1)(s_20)}\Big\}\notag\\
  &=\ee^{\Lc_S\tau}\!\int_{\tau>s_1>s_2>0}\Big\{W_{\text{crossed}}+W_{\text{nested}}\Big\},
\end{align}
exactly the two irreducible fourth-order diagrams of \cref{fig:fourth-order}. The
subtraction of the reducible pairing is not an approximation imposed by hand; it is the
single $\Pc$ that survives in the middle gap. For $\theta=0$ on the coherence, with all
vertices equal to $-2i$ and every line a noise line, this is
$16\int_{\tau>s_1>s_2>0}[S(\tau-s_2)S(s_1)+S(\tau)S(s_1-s_2)]$, the crossed and nested integrals
of \cref{eq:dephasing-4th}.

The identity can be checked numerically to the letter. In the pseudomode representation
(\cref{sec:numerics}) both $K(\tau)$, as $\Pc\Lc_I\ee^{\Qc\Lc\tau}\Qc\Lc_I\Pc$, and $K^{(4)}(\tau)$, as the
double integral of $\Pc\Lc_I\ee^{\Lc_0\cdot}\Qc\Lc_I\ee^{\Lc_0\cdot}\Qc\Lc_I\ee^{\Lc_0\cdot}\Qc\Lc_I\Pc$, are
finite matrices. For the Lorentzian bath at $\theta=\pi/4$ one finds, e.g.\ at $\tau=1/\omega_0$,
$\max|K-K^{(2)}|=4.1\times10^{-4}$ and $\max|K-K^{(2)}-K^{(4)}|=4\times10^{-6}$ at $g=0.1$, and
$2.6\times10^{-5}$ versus $6\times10^{-7}$ at $g=0.05$: the residual is of order $g^6$ (and
dominated by the quadrature error), as it must be.
\end{exbox}

\begin{exbox}[Example 5.2: sixth order, and the count of irreducible diagrams]
With $m=4$ there are five gaps, numbered $0$ to $4$ from the latest vertex. The parity
argument kills every $\mathcal C$ that contains gap $0$, $2$ or $4$ (a $\Pc$ there leaves an
odd number of vertices on one side). The surviving sets are $\varnothing$, $\{1\}$, $\{3\}$ and
$\{1,3\}$:
\begin{equation}
  K^{(6)}=\ee^{\Lc_S\tau}\!\int\Big\{W_6-W_2W_4-W_4W_2+W_2W_2W_2\Big\},
\end{equation}
where $W_6$ is the full six-vertex string ($15$ pairings), $W_2W_4$ the pairings with a cut
after the first two vertices ($1\times3=3$ pairings), $W_4W_2$ those with a cut before the
last two ($3$ pairings), and $W_2W_2W_2$ the single pairing with both cuts. Counting:
$15-3-3+1=10$ irreducible pairings, which matches the sequence $1,2,10,74,\dots$ of
irreducible chord diagrams quoted in \cref{ex:4-3}. The reducible ones are $5=15-10$:
the pairing with two cuts, and the $2+2$ pairings with exactly one cut (an irreducible
fourth-order block next to a second-order one, as in \cref{fig:cut}). Note that the
inclusion--exclusion signs are needed to avoid double-subtracting the pairing with two
cuts, which appears in both $W_2W_4$ and $W_4W_2$.
\end{exbox}

\subsubsection{What the theorem does and does not say}

\begin{itemize}[itemsep=2pt]
\item \textbf{Irreducibility is relative to $\Pc$.} A cut is a moment at which the bath
  is \emph{exactly} back in the reference state $\rhoB$. Change the reference state (a
  correlated projector, a projector onto populations only as in \cref{sec:lec7}, Mori's
  operator projector of \cref{sec:lec8}) and both the ``free'' propagation and the set
  of diagrams that count as reducible change with it. The kernel is not a property of
  the model alone but of the model and the projector.
\item \textbf{$K(0)=K^{(2)}(0)$ exactly.} At $\tau=0$ there is no room for internal vertices,
  so only $m=0$ contributes. This fixes the short-time behaviour of every quantity
  to all orders (used in \cref{sec:lec10}).
\item \textbf{$K$ is small; $\Pi$ is not.} $K$ starts at order $g^2$ while $\Pi$ starts at
  order $1$. Truncating $K$ and then solving \cref{eq:NZ} resums an infinite class of
  diagrams of $\Pi$: all products of the kept irreducible blocks. This is why a
  second-order kernel can describe long-time exponential decay, which no finite-order
  truncation of $\Pi$ itself can (a polynomial in $t$ does not decay).
\item \textbf{Truncating $K$ is not the same as truncating $\Gc$.} The TCL generator of
  \cref{sec:lec6} resums a \emph{different} class of diagrams at the same nominal order;
  the two agree at second order and differ from fourth order on~\cite{Timm2008,LeijnseWegewijs2008}.
\item \textbf{The theorem is representation-independent.} It is the real-time, Liouville-space
  version of ``the self-energy is one-particle irreducible''. In the fermionic
  reservoir context the identification is due to Schoeller and
  Sch\"on~\cite{SchoellerSchon1994,Schoeller2000}; it is the starting point of the
  real-time renormalization group~\cite{Schoeller2009,Keil2001}, and the explicit
  fourth-order comparison with the NZ expansion is in~\cite{LeijnseWegewijs2008,Timm2008}.
  Historically, Zwanzig~\cite{Zwanzig1964} already showed that the projection-operator
  master equation coincides with the resummed perturbation series of Prigogine's school
  and of van Hove; the present diagrammatic form is the modern version of that
  identity.
\end{itemize}

\subsection{Second order: Born, Redfield, secular, Bloch}
\label{sec:born}

At order $\Lc_I^2$ we drop $\Qc$ in \cref{eq:NZ} (because $\Pc\Lc_I\Pc=0$) and replace
$\ee^{\Qc\Lc\tau}\to\ee^{\Lc_0\tau}$; the result is $K^{(2)}$ of \cref{eq:K2}, now identified as
the first term of the irreducible series. The approximations one layers on top are:
\begin{enumerate}[itemsep=1pt]
\item \textbf{Born (NZ2).} Keep $K\approx K^{(2)}$ in \cref{eq:NZ}, as a time-nonlocal equation.
\item \textbf{Redfield (time-local, nonsecular).} Replace $\rhoS(s)\to\ee^{-\Lc_S(t-s)}\rhoS(t)$
  under the integral (valid when $\rhoS$ changes little, in the interaction picture, over
  the bath correlation time) and extend the upper limit to $\infty$:
  \begin{equation}
    \dot\rhoS=\big[\Lc_S+\Gc^{(2)}_\infty\big]\rhoS,\qquad
    \Gc^{(2)}_\infty=\int_0^\infty\!\dd\tau\,K^{(2)}(\tau)\,\ee^{-\Lc_S\tau}.
    \label{eq:redfield}
  \end{equation}
  This is Redfield's equation~\cite{Redfield1957,WangsnessBloch1953,Bloch1957}; we will
  see in \cref{sec:lec6} that it is the $t\to\infty$ limit of the second-order TCL generator.
\item \textbf{Secular.} Drop the terms of $\Gc^{(2)}_\infty$ that oscillate at $\pm\omega_0$ or $\pm2\omega_0$ in
  the interaction picture. The result is of GKSL form \eqref{eq:GKSL} and generates a
  completely positive semigroup (Davies' theorem~\cite{Davies1974}; see
  \cite[Sec.~3.3.1]{BreuerPetruccione} for the general construction). Without the secular
  step, Redfield can violate positivity at short times.
\end{enumerate}

\begin{exbox}[Example 5.3: the Bloch equations for the qubit]
Apply \cref{eq:redfield} to the kernel \eqref{eq:K2-bloch}. The rotation $\ee^{-\Lc_S\tau}$ acts
on the \emph{input}: $(x_x,x_y)\mapsto(\cos\,x_x+\sin\,x_y,\,-\sin\,x_x+\cos\,x_y)$. Multiplying out
and keeping the secular terms (the $\tau$-integrals of products such as
$S_\tau\cos\omega_0\tau\cos\omega_0\tau$ contain a non-oscillating half; the $cs$ cross terms and
the $\cos^2$-type oscillating halves drop), one finds the Bloch equations
\begin{equation}
  \dot r_z=-\Gamma_1\big(r_z-r_z^{\rm eq}\big),\qquad
  \dot r_\pm=\big(\pm i\tilde\omega_0-\Gamma_2\big)r_\pm,\qquad r_\pm\equiv r_x\pm ir_y ,
  \label{eq:bloch-eqs}
\end{equation}
with
\begin{align}
  \Gamma_1&=\Gamma_\downarrow+\Gamma_\uparrow=2\sin^2\theta\,S(\omega_0),\qquad
  \Gamma_{\downarrow}=2\sin^2\theta\,J(\omega_0)\big[1+n(\omega_0)\big],\quad
  \Gamma_{\uparrow}=2\sin^2\theta\,J(\omega_0)\,n(\omega_0),
  \label{eq:rates-T1}\\
  r_z^{\rm eq}&=-\frac{\chi''(\omega_0)}{S(\omega_0)}=-\tanh\frac{\beta\omega_0}{2},
  \label{eq:rz-eq}\\
  \Gamma_2&=\tfrac12\Gamma_1+\Gamma_\varphi,\qquad \Gamma_\varphi=2\cos^2\theta\,S(\omega{=}0),
  \label{eq:rates-T2}\\
  \tilde\omega_0&=\omega_0+\delta\omega,\qquad
  \delta\omega=2\sin^2\theta\int_0^\infty\!\dd\tau\,S(\tau)\sin\omega_0\tau
  =\frac{\sin^2\theta}{\pi}\,\mathrm P\!\!\int_{-\infty}^{\infty}\!\dd\omega\,\frac{S(\omega)}{\omega_0-\omega}.
  \label{eq:lamb}
\end{align}
Each line is worth a comment.
\begin{itemize}[itemsep=1pt]
\item \Cref{eq:rates-T1}: the population column of $K^{(2)}$ gives
  $\int_0^\infty K_{zz}\dd\tau=-4\sin^2\theta\int_0^\infty S(\tau)\cos\omega_0\tau\,\dd\tau=-2\sin^2\theta S(\omega_0)$. The
  \emph{sum} of the rates is set by the noise at $\omega_0$. Using the FDT,
  $S(\omega_0)=J(\omega_0)[1+2n(\omega_0)]$, which splits into $\Gamma_\downarrow+\Gamma_\uparrow$ as written. This is
  Fermi's golden rule with the $C(\pm\omega_0)$ of \cref{eq:FDT}.
\item \Cref{eq:rz-eq}: the inhomogeneous term is
  $\int_0^\infty K_{z\Id}\dd\tau=-2\sin^2\theta\int_0^\infty\chi(\tau)\sin\omega_0\tau\,\dd\tau=-2\sin^2\theta\,\chi''(\omega_0)$. The
  \emph{difference} of the rates, $\Gamma_\downarrow-\Gamma_\uparrow=2\sin^2\theta\chi''(\omega_0)$, is set by the
  response. The ratio $\chi''/S=\tanh(\beta\omega_0/2)$ is the FDT, and it makes the steady state
  the Gibbs state of $H_S$: detailed balance is the fluctuation--dissipation theorem seen
  from the system. Out of equilibrium (a driven or nonthermal bath), $S$ and $\chi''$ are
  independent and $r_z^{\rm eq}$ defines an effective temperature.
\item \Cref{eq:rates-T2}: pure dephasing samples the noise at zero frequency and
  comes with $\cos^2\theta$; relaxation contributes half its rate to the coherence decay. For
  $S=\sz/2$ one would have $\Gamma_\varphi=\tfrac12\cos^2\theta S(0)$.
\item \Cref{eq:lamb}: the Lamb shift is the Hilbert transform of the \emph{noise} spectrum.
  At $T=0$, $S(\omega)=J(|\omega|)$ and the integral splits into the rotating-wave term
  $\mathrm P\!\int_0^\infty J(\omega)/(\omega_0-\omega)$ and the counter-rotating (Bloch--Siegert) term
  $\int_0^\infty J(\omega)/(\omega_0+\omega)$. The response $\chi'$ drops out of the shift because it is
  even in $\omega$ (\cref{ex:5-2}).
\end{itemize}
For the Lorentzian bath of \cref{sec:lec2} at $T=0$, with $\Omega=\omega_0=1$, $\gamma=0.5$,
$\theta=\pi/4$, $g=0.3$: $\Gamma_\downarrow=0.360$, $\Gamma_\uparrow=0.0055$ (from the negative-frequency
tail), $\Gamma_\varphi=0.0424$, $\Gamma_2=0.225$, $\delta\omega=g^2\sin^2\theta\,\frac{2\Omega}{4\Omega^2+\gamma^2/4}=0.022$. The
script of \cref{sec:numerics} computes $\Gc^{(2)}_\infty$ numerically and reproduces all of these
to three digits (the nonsecular generator gives a precession frequency $1.0219$ versus
$1.0222$ secular).
\end{exbox}

\centering
\includegraphics[width=\textwidth]{figures/fig_dynamics.pdf}

\Cref{fig:dynamics} compares the three second-order schemes with the exact solution.
Three lessons: (i) at weak coupling all agree, and the Markovian rates of Example~5.3 are
quantitatively right after a transient of duration $\sim1/\gamma$; (ii) the Born kernel used
\emph{nonlocally} (NZ2) is the \emph{worst} of the three at stronger coupling, overshooting
to unphysical states, a well-known pathology~\cite{SmirneVacchini2010}; (iii) the exact
steady state is not the Gibbs state of $H_S$ but the reduced state of the global
equilibrium, which the second-order schemes can only approximate.

\subsection{Beyond second order}

Example~5.1 gives the fourth-order kernel as the crossed and nested diagrams of
\cref{fig:fourth-order}, each with its full vertex structure (all assignments of
$\cl/\qu$ consistent with the rules, i.e.\ $\qu\qu$ lines carrying $S$ and $\qu\cl$ lines
carrying $-i\chi$). In the Schr\"odinger picture, with $V_t\in\{-iS^\qu,-iS^\cl\}$ the vertex at
time $t$ and free propagators $\ee^{\Lc_S\Delta t}$ between vertices,
\begin{equation}
  K^{(4)}(\tau)=\int_{\tau>t_1>t_2>0}\Big\{V_\tau\,\ee^{\Lc_S(\tau-t_1)}V_{t_1}\ee^{\Lc_S(t_1-t_2)}V_{t_2}\ee^{\Lc_St_2}V_0\Big\}_{\text{crossed }(\tau t_2)(t_10)\ +\ \text{nested }(\tau0)(t_1t_2)} .
\end{equation}
For $\theta=0$ on the coherence it reduces to the crossed and nested integrals of
\cref{eq:dephasing-4th} times $16$. Explicit fourth-order kernels for the spin-boson model
were worked out in the TCL form by Breuer, Kappler and
Petruccione~\cite{BreuerKapplerPetruccione2001}; in the NZ form the bookkeeping is easiest
in Laplace space~\cite{Timm2008}. \Cref{fig:kernels} (left) shows the exact kernel of the
Lorentzian model, extracted numerically from the pseudomode representation as
$\Pc\Lc_I\ee^{\Qc\Lc\tau}\Qc\Lc_I\Pc$ (which is \cref{eq:NZ} evaluated literally, with the
pseudomode's Markovian Liouvillian playing the role of $\Lc$), against $K^{(2)}$: the higher
orders shift the oscillation frequency inside the kernel (a self-consistent Lamb shift)
and change its damping, but $K(0)=K^{(2)}(0)$ exactly, as \cref{thm:irreducible} requires.

\subsection{Correlated initial states and the inhomogeneity}

If $\rho(0)$ is not factorized, $I(t)$ in \cref{eq:NZ} is nonzero. It decays on the bath
correlation time if the initial correlations are short-ranged, and it can be absorbed
into a modified kernel in some cases. In the Keldysh language the same information lives
on the imaginary-time (Matsubara) branch attached to the contour at
$t=0$~\cite[Ch.~4]{StefanucciVanLeeuwen}. One should not expect $\Pi(t)$ to be a
completely positive map of $\rhoS(0)$ in that case~\cite{Pechukas1994}.

\subsection{Exercises}

\begin{exercise}\label{ex:5-1}
Carry out the derivation of \cref{eq:NZ} with a general projector $\Pc$ (not necessarily
of the form \eqref{eq:P}) and identify which steps use $\Pc\Lc_I\Pc=0$ and $[\Pc,\Lc_0]=0$. What
changes if $\avg{B}\ne0$? (Hint: absorb $S\avg{B}$ into $H_S$.)
\end{exercise}

\begin{exercise}\label{ex:5-2}
Derive the Bloch equations \eqref{eq:bloch-eqs}--\eqref{eq:lamb} from \cref{eq:K2-bloch}.
Show that the real part of the response, $\chi'$, cancels from $\delta\omega$ and explain why
with a parity argument. Then obtain the same rates from the operator form
$\Gc X=-i[H_S+H_{\rm LS},X]+\sum_\omega\gamma(\omega)\big(A(\omega)XA(\omega)^\dagger-\tfrac12\{A(\omega)^\dagger A(\omega),X\}\big)$ with
$A(\pm\omega_0)=\sin\theta\,\sigma_\mp$, $A(0)=\cos\theta\,\sz$, $\gamma(\omega)=C(\omega)$~\cite[Sec.~3.3.1]{BreuerPetruccione}.
\end{exercise}

\begin{exercise}\label{ex:5-3}
Show that the secular generator of \cref{ex:5-2} is of GKSL form with
$L_1=\sigma_-$, $L_2=\sigma_+$, $L_3=\sz$ and nonnegative rates. Then take the nonsecular
Redfield generator for $\theta=\pi/2$ and find an initial state and a time at which
$\Tr\rhoS^2>1$ for a Lorentzian bath with $\gamma\gtrsim\omega_0$.
\end{exercise}

\begin{exercise}\label{ex:5-4}
Prove $K(0)=K^{(2)}(0)$ directly from \cref{eq:NZ}. Then show that in Laplace space
$\hat K(z)=z-\Lc_S-\hat\Pi(z)^{-1}$, so that the exact kernel can be \emph{extracted} from the
exact map (this is the route of Shi and Geva~\cite{ShiGeva2003} and of Cohen and
Rabani~\cite{CohenRabani2011}). Use it to show that for pure dephasing ($\theta=0$) the
coherence kernel $\hat K_{01}(z)$ is an infinite series in $g^2$ even though the exact
solution \eqref{eq:dephasing-exact} is ``second order'' in a different sense.
\end{exercise}
